// ============================================================ // CONFIGURACIÓN // ============================================================ const SHEET_ID = '1-bkFZxe7UuzJwNTqLjEnAEr4QWMMCMgl_UV_84rpve8'; const SHEET_NAME = 'Negocios'; const FOLDER_NAME = 'Fotos Negocios (App)'; // Encabezados canónicos de la Hoja, en este orden exacto. const HEADERS = [ 'IdNegocio', 'Usuario', 'Password', 'Nombre', 'Slogan', 'Descripcion', 'Ubicacion', 'Direccion', 'UrlFacebook', 'Telefono', 'Whatsapp', 'Correo', 'Horario', 'Rol', 'Activo', 'UltimoAcceso', 'Foto1', 'Desc1', 'Foto2', 'Desc2', 'Foto3', 'Desc3', 'Foto4', 'Desc4' ]; // ============================================================ // ENTRADA // ============================================================ function doGet() { return HtmlService .createHtmlOutputFromFile('Admin') .setTitle('Gestión de Negocios') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } /** * Ejecutar UNA VEZ manualmente desde el editor de Apps Script * para crear la pestaña "Negocios" con los encabezados correctos * si la Hoja está vacía o no existe la pestaña. */ function inicializarHoja() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } return 'Listo'; } // ============================================================ // UTILIDADES DE MAPEO FILA <-> OBJETO // ============================================================ function getSheet_() { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); if (!sheet) throw new Error('No existe la pestaña "' + SHEET_NAME + '". Ejecuta inicializarHoja() primero.'); return sheet; } function rowToNegocio_(rowArray) { const obj = {}; HEADERS.forEach(function (h, i) { obj[h] = rowArray[i]; }); const fotos = []; for (let n = 1; n <= 4; n++) { const url = obj['Foto' + n]; if (url) fotos.push({ url: url, descripcion: obj['Desc' + n] || '' }); } return { id: obj.IdNegocio, usuario: obj.Usuario, password: obj.Password, nombre: obj.Nombre, slogan: obj.Slogan, descripcion: obj.Descripcion, ubicacion: obj.Ubicacion, direccion: obj.Direccion, urlFacebook: obj.UrlFacebook, telefono: obj.Telefono, whatsapp: obj.Whatsapp, correo: obj.Correo, horario: obj.Horario, rol: obj.Rol, activo: (obj.Activo === true || obj.Activo === 'TRUE' || obj.Activo === 'true'), ultimoAcceso: obj.UltimoAcceso, fotos: fotos }; } function negocioToRow_(datos) { const map = { IdNegocio: datos.id || '', Usuario: datos.usuario || '', Password: datos.password || '', Nombre: datos.nombre || '', Slogan: datos.slogan || '', Descripcion: datos.descripcion || '', Ubicacion: datos.ubicacion || '', Direccion: datos.direccion || '', UrlFacebook: datos.urlFacebook || '', Telefono: datos.telefono || '', Whatsapp: datos.whatsapp || '', Correo: datos.correo || '', Horario: datos.horario || '', Rol: datos.rol || '', Activo: datos.activo === true, UltimoAcceso: new Date().toLocaleString('es-MX') }; for (let n = 1; n <= 4; n++) { const foto = (datos.fotos && datos.fotos[n - 1]) || {}; map['Foto' + n] = foto.url || ''; map['Desc' + n] = foto.descripcion || ''; } return HEADERS.map(function (h) { return map[h] !== undefined ? map[h] : ''; }); } function findRowIndexById_(sheet, id) { const data = sheet.getDataRange().getValues(); const idCol = HEADERS.indexOf('IdNegocio'); for (let r = 1; r < data.length; r++) { if (data[r][idCol] === id) return r + 1; // 1-indexado, +1 por encabezado } return -1; } function siguienteId_(sheet) { const data = sheet.getDataRange().getValues(); const idCol = HEADERS.indexOf('IdNegocio'); let max = 0; for (let r = 1; r < data.length; r++) { const val = String(data[r][idCol] || ''); const match = val.match(/USR-(\d+)/); if (match) max = Math.max(max, parseInt(match[1], 10)); } return 'USR-' + String(max + 1).padStart(3, '0'); } // ============================================================ // CRUD // ============================================================ function obtenerNegocios() { try { const sheet = getSheet_(); const data = sheet.getDataRange().getValues(); const negocios = []; for (let r = 1; r < data.length; r++) { if (data[r].join('') === '') continue; // fila vacía negocios.push(rowToNegocio_(data[r])); } return { success: true, negocios: negocios }; } catch (e) { return { success: false, error: e.message }; } } function agregarNegocio(datos) { try { const sheet = getSheet_(); if (!datos.id) datos.id = siguienteId_(sheet); sheet.appendRow(negocioToRow_(datos)); return { success: true, id: datos.id }; } catch (e) { return { success: false, error: e.message }; } } function editarNegocio(idNegocio, datos) { try { const sheet = getSheet_(); const rowIndex = findRowIndexById_(sheet, idNegocio); if (rowIndex === -1) return { success: false, error: 'Negocio no encontrado' }; datos.id = idNegocio; // Si no llegan fotos nuevas, conservar las existentes if (!datos.fotos) { const existente = rowToNegocio_(sheet.getRange(rowIndex, 1, 1, HEADERS.length).getValues()[0]); datos.fotos = existente.fotos; } sheet.getRange(rowIndex, 1, 1, HEADERS.length).setValues([negocioToRow_(datos)]); return { success: true }; } catch (e) { return { success: false, error: e.message }; } } function eliminarNegocio(idNegocio) { try { const sheet = getSheet_(); const rowIndex = findRowIndexById_(sheet, idNegocio); if (rowIndex === -1) return { success: false, error: 'Negocio no encontrado' }; sheet.deleteRow(rowIndex); return { success: true }; } catch (e) { return { success: false, error: e.message }; } } // ============================================================ // FOTOS (Google Drive) // ============================================================ function getOrCreateFolder_() { const folders = DriveApp.getFoldersByName(FOLDER_NAME); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(FOLDER_NAME); } /** * Sube una foto en base64 (dataURL completo o solo el payload) a Drive * y regresa una URL pública de solo-lectura. */ function subirFoto(base64Data, nombreArchivo, mimeType) { try { const folder = getOrCreateFolder_(); const payload = base64Data.indexOf(',') > -1 ? base64Data.split(',')[1] : base64Data; const bytes = Utilities.base64Decode(payload); const blob = Utilities.newBlob(bytes, mimeType || 'image/jpeg', nombreArchivo || 'foto.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const url = 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1000'; return { success: true, url: url, fileId: file.getId() }; } catch (e) { return { success: false, error: e.message }; } } // ============================================================ // ESTADÍSTICAS // ============================================================ function obtenerEstadisticas() { const resultado = obtenerNegocios(); if (!resultado.success) return { success: false, error: resultado.error }; const negocios = resultado.negocios; const activos = negocios.filter(function (n) { return n.activo === true; }).length; const inactivos = negocios.length - activos; const ubicaciones = new Set(); negocios.forEach(function (n) { if (n.ubicacion) ubicaciones.add(n.ubicacion); }); return { success: true, estadisticas: { total: negocios.length, activos: activos, inactivos: inactivos, ubicacionesUnicas: ubicaciones.size } }; }
